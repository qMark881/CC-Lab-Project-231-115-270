```c
int x;
int y;
float f;
bool b;

// Integer operations
x = 10;
y = 20;
x = x + y;
print x;

// Float operations
f = 3.14;
f = f * 2.0;
print f;

// Boolean operations
b = true;
b = !b;
print b;

// Relational operations
x = 10;
y = 20;
b = x < y;
print b;

b = x > y;
print b;

b = x == 10;
print b;

b = y != 20;
print b;

// Mixed type comparisons
f = 10.5;
b = x < f;
print b;

// Complex expressions
x = (10 + 20) * 2;
print x;

f = (3.14 + 2.86) / 2.0;
print f;

b = (x > 0) && (f > 0);
print b;
```
