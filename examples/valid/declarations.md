```c
// Variable declarations with initialization
int x = 10;
int y = 20;
float f = 3.14;
bool b = true;

// Use the initialized variables
print x;
print y;
print f;
print b;

// Declaration without initialization
int z;
z = 30;
print z;

// Multiple declarations
int a = 5;
int b = 10;
int c = 15;

print a;
print b;
print c;

// Reassignment
x = 100;
y = 200;
f = 6.28;
b = false;

print x;
print y;
print f;
print b;
```
