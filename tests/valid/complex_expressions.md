```c
int x;
int y;
int z;
float a;
bool b;

x = 10;
y = 20;
z = 30;
a = 3.14;
b = true;

// Complex arithmetic expressions
x = (x + y) * z;
y = x / (y + 1);
z = x % y;

// Complex logical expressions
b = (x > y) && (z < x);
b = (x == 10) || (y != 20);
b = !((x > 0) && (y < 100));

// Mixed expressions
x = (int)a;
y = x + (int)a;
```
